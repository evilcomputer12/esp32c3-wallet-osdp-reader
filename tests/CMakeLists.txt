cmake_minimum_required(VERSION 3.16)
project(reader_checks C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_compile_options(-g -fsanitize=address,undefined -fno-omit-frame-pointer)
add_link_options(-fsanitize=address,undefined)
set(ROOT ${CMAKE_CURRENT_SOURCE_DIR}/..)
set(WOLF ${ROOT}/dependencies/wolfssl)
add_library(wolf STATIC)
foreach(file aes sha256 hash random wc_port wc_encrypt error memory logging ecc sp_int sp_c32 sp_c64 asn coding wolfmath hmac kdf)
    target_sources(wolf PRIVATE ${WOLF}/wolfcrypt/src/${file}.c)
endforeach()
target_include_directories(wolf PUBLIC ${ROOT}/include ${WOLF})
target_compile_definitions(wolf PUBLIC WOLFSSL_USER_SETTINGS READER_HOST_CRYPTO_TEST)
include(CheckIncludeFile)
check_include_file(strings.h READER_HAVE_STRINGS_H)
if(READER_HAVE_STRINGS_H)
    # Linux strict C11 declares strcasecmp in strings.h, not string.h.
    target_compile_definitions(wolf PUBLIC HAVE_STRINGS_H)
endif()
add_library(wallet STATIC ${ROOT}/src/wallet_reader.c ${ROOT}/src/wallet_crypto.c)
target_link_libraries(wallet PUBLIC wolf)
target_compile_options(wallet PRIVATE -Wall -Wextra -Werror)
add_executable(wallet_driver wallet_driver.c)
target_link_libraries(wallet_driver PRIVATE wallet)
enable_testing()
add_executable(test_wallet test_wallet.c ${ROOT}/src/crypto_selftest.c)
target_link_libraries(test_wallet PRIVATE wallet)
target_compile_options(test_wallet PRIVATE -Wall -Wextra -Werror -Wpedantic)
add_test(NAME wallet COMMAND test_wallet)
add_executable(test_pn532 test_pn532.c ${ROOT}/src/pn532.c)
target_include_directories(test_pn532 PRIVATE ${ROOT}/include)
target_compile_options(test_pn532 PRIVATE -Wall -Wextra -Werror)
add_test(NAME pn532 COMMAND test_pn532)
add_executable(test_policy test_policy.c)
target_include_directories(test_policy PRIVATE ${ROOT}/include)
target_compile_options(test_policy PRIVATE -Wall -Wextra -Werror)
add_test(NAME policy COMMAND test_policy)

# Exercise the production wolfCrypt OSDP binding, including ACU/PD loopback.
set(OSDP ${ROOT}/dependencies/osdp-embedded)
file(GLOB_RECURSE OSDP_SRC ${OSDP}/core/src/*.c ${OSDP}/pd/src/*.c ${OSDP}/acu/src/*.c)
add_library(osdp_host STATIC ${OSDP_SRC})
target_include_directories(osdp_host PUBLIC ${OSDP}/core/include ${OSDP}/pd/include ${OSDP}/acu/include
    PRIVATE ${OSDP}/core/src ${OSDP}/pd/src ${OSDP}/acu/src)
target_compile_definitions(osdp_host PUBLIC OSDP_PD_BUF_LEN=256 OSDP_STREAM_BUFFER_LEN=1440)
add_library(unity STATIC ${OSDP}/tests/vendor/unity/unity.c)
add_executable(test_feedback test_feedback.c ${ROOT}/src/feedback.c)
target_include_directories(test_feedback PRIVATE ${ROOT}/include ${CMAKE_CURRENT_SOURCE_DIR}/stubs)
target_link_libraries(test_feedback PRIVATE osdp_host)
target_compile_options(test_feedback PRIVATE -Wall -Wextra -Werror -Wpedantic)
add_test(NAME feedback COMMAND test_feedback)
target_include_directories(unity PUBLIC ${OSDP}/tests/vendor/unity)
add_library(osdp_wolf_port STATIC ${OSDP}/ports/wolfcrypt/osdp_sc_wolfcrypt.c)
target_include_directories(osdp_wolf_port PUBLIC ${OSDP}/ports/wolfcrypt)
target_link_libraries(osdp_wolf_port PUBLIC osdp_host wolf)
add_library(sc_adapter STATIC ${OSDP}/tests/sc_test_aes.c)
target_compile_definitions(sc_adapter PRIVATE OSDP_SC_TEST_WOLFCRYPT)
target_include_directories(sc_adapter PUBLIC ${OSDP}/tests)
target_link_libraries(sc_adapter PUBLIC osdp_wolf_port)
foreach(name test_sc_aes test_sc_keys test_sc_cbc test_sc_mac test_sc_payload test_sc_wrap test_pd_sc test_acu_sc test_loopback_sc test_capture_replay test_port_wolfcrypt)
    add_executable(${name} ${OSDP}/tests/${name}.c)
    target_link_libraries(${name} PRIVATE sc_adapter unity)
    target_include_directories(${name} PRIVATE ${OSDP}/core/src)
    add_test(NAME ${name}_wolfcrypt COMMAND ${name})
endforeach()
